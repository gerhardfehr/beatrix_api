\section{K-Metric Mathematics}

\subsection{Formulas}

\textbf{Standard K-Metric:}
\begin{equation}
K = 1 - \frac{|B - E|}{\max(B, E)} = \frac{\min(B, E)}{\max(B, E)}
\end{equation}

\textbf{Partial Derivatives:}
\begin{align}
\frac{\partial K}{\partial B} &= \begin{cases} E/B^2 & \text{if } B > E \\ 1/E & \text{if } B \leq E \end{cases} \\
\frac{\partial K}{\partial E} &= \begin{cases} 1/B & \text{if } B > E \\ B/E^2 & \text{if } B \leq E \end{cases}
\end{align}

\textbf{Error Propagation:}
\begin{equation}
\sigma_K = \sqrt{\left(\frac{\partial K}{\partial B}\right)^2 \sigma_B^2 + \left(\frac{\partial K}{\partial E}\right)^2 \sigma_E^2}
\end{equation}

\textbf{Monte-Carlo Consistency:}
\begin{equation}
\kappa_{\text{MC}} = 1 - \frac{\sigma_K}{\sigma_{\max}} \quad \text{where } \sigma_{\max} = \frac{1}{\sqrt{12}} \approx 0.289
\end{equation}

\begin{enumerate}
\item \textbf{CTC} (Claim-Type Classification): $E_{\text{eff}} = \min(E, E_{\max}(T))$

\item \textbf{EVA} (External Validation): $K \leq 0.45$ for unvalidated T4/T5
\item \textbf{RAS} (Reflexive Application Stop): Definitional claims blocked
\end{enumerate}

\section{Glossary}

\begin{description}
\item[B-value] Boldness value; claim strength from linguistic markers. Range [0,1].
\item[Calibration] Alignment between claim confidence and evidence.
\item[CTC] Claim-Type Classification; $E_{\max}$ ceiling safeguard.
\item[E-value] Evidence value from discipline hierarchy. Range [0,1].
\item[ESL] Empirical Stability of Language framework.
\item[EVA] External Validation Requirement; caps T4/T5 at K=0.45.
\item[Highly Stable] Status for $K \geq 0.85$.
\item[K-metric] $K = \min(B,E)/\max(B,E)$. Range [0,1].
\item[$K_{\text{stable}}$] Discipline-specific stability threshold.
\item[$\kappa_{\text{MC}}$] Monte-Carlo consistency: $1 - \sigma_K/\sigma_{\max}$.
\item[Overclaiming] $B > E$; confidence exceeds evidence.
\item[Partially Stable] Status for $K \in [0.50, 0.75)$.
\item[PP] Provenance Protocol; source modifier safeguard.
\item[RAS] Reflexive Application Stop; blocks self-validation.
\item[Stable] Status for $K \in [0.75, 0.85)$.
\item[Underclaiming] $B < E$; confidence below evidence.
\item[Unstable] Status for $K < 0.50$.
\item[WLA] Weakest-Link Aggregation; $K_{\text{chain}} = \min(K_i)$.
\end{description}
