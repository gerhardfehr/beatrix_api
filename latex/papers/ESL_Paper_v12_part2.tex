\section{Scope Conditions: Where Does ESL Apply?}
\label{sec:scope}

\begin{analogybox}[The Jurisdiction of a Court]
A court has jurisdiction over specific types of cases. A traffic court handles parking tickets, not murder trials. ESL similarly has ``jurisdiction'' over \textbf{assertive, literal, informative claims where speakers assume epistemic responsibility}. Poetry is not in ESL's jurisdiction. Neither is fiction, or small talk.
\end{analogybox}


\subsection{The Four Necessary Conditions}

ESL applies if and only if all four conditions are satisfied:

\begin{principlebox}[ESL Applicability Conditions]
\textbf{C1: Truth Claim} --- The utterance purports to assert something true about the world

\textbf{C2: Literal Meaning} --- The utterance is meant literally, not figuratively

\textbf{C3: Informative Function} --- The primary function is to convey information

\textbf{C4: Epistemic Responsibility} --- The speaker assumes responsibility for the truth of the claim
\end{principlebox}

\subsubsection{Poetic and Aesthetic Language}
Poetry violates C2 (literal meaning) and often C1 (truth claim in the empirical sense).

\textbf{Example:} ``All happy families are alike; each unhappy family is unhappy in its own way.'' --- Tolstoy

This is a \emph{literary truth}---not an empirical claim about families.

\subsubsection{Fictional Language}
Fiction systematically violates C1 (truth claim). ``One morning, Gregor Samsa woke from troubled dreams to find himself transformed into a monstrous insect.'' is not overclaiming despite $K = 1.0$ and $M = 0.0$.

\section{Multi-Level Aggregation: From Sentence to Text}
\label{sec:aggregation}

\begin{analogybox}[The Financial Audit]
When auditing finances, you don't just check individual transactions. You aggregate from transactions $\rightarrow$ departments $\rightarrow$ divisions $\rightarrow$ company-wide.

\textbf{Key insight:} The maximum matters more than the mean. One severely overclaiming sentence in the abstract ($\Delta = +0.50$) damages credibility more than ten mildly overclaiming sentences in the methods.
\end{analogybox}


\subsection{Five Aggregation Rules}

\paragraph{Rule 1: Arithmetic Mean}
$\Delta(T) = \frac{1}{n} \sum_{i=1}^{n} \Delta_i$

\paragraph{Rule 2: Maximum Rule}
$\Delta(T) = \max(\Delta_1, \Delta_2, \ldots, \Delta_n)$

\paragraph{Rule 3: Overclaiming Rate}
$\text{OC-Rate} = \frac{|\{i : \Delta_i > 0.25\}|}{n}$

\paragraph{Rule 4: Position-Weighted}
$\Delta(T) = \sum_{i=1}^{n} w_i \cdot \Delta_i$

\paragraph{Rule 5: Hierarchical (Core vs. Periphery)}
Distinguish between core claims and peripheral claims.


\subsection{The Integrated Multi-Metric Framework}

\begin{table}[H]
\centering
\caption{The seven ESL text metrics with recommended thresholds}
\begin{tabular}{llc}
\toprule
\textbf{Metric} & \textbf{Definition} & \textbf{Threshold} \\
\midrule
$\Delta_{\text{mean}}$ & Mean of all $\Delta_i$ & $\leq 0.10$ \\
$\Delta_{\text{max}}$ & Maximum $\Delta_i$ & $\leq 0.50$ \\
OC-Rate & Proportion with $\Delta > 0.25$ & $\leq 5\%$ \\
$\Delta_{\text{weighted}}$ & Position-weighted mean & $\leq 0.10$ \\
$\Delta_{\text{abstract}}$ & Abstract only & $\leq 0.05$ \\
$\Delta_{\text{title}}$ & Title only & $\leq 0.00$ \\

\end{tabular}
\end{table}

\textbf{Decision Tree:}
\begin{enumerate}
    \item If $\Delta_{\text{core}} > 0.25$ $\rightarrow$ \textbf{CRITICAL}
    \item Else if OC-Rate $> 15\%$ $\rightarrow$ \textbf{SYSTEMATIC}
    \item Else if $\Delta_{\text{max}} > 0.50$ $\rightarrow$ \textbf{WARNING}
    \item Else if $\Delta_{\text{weighted}} > 0.10$ $\rightarrow$ \textbf{MINOR}
    \item Else $\rightarrow$ \textbf{ALIGNED}
\end{enumerate}

\section{Compositional Semantics: How Elements Modify Each Other}

\begin{analogybox}[The Recipe]
The final dish is not just the sum of ingredients---it's how they combine. Language works similarly. ``X causes Y'' has $K = 0.90$. Add ``might'' and it becomes $K = 0.45$. Add ``in some cases'' and $K \approx 0.20$. The words \emph{multiply}, they don't simply add.
\end{analogybox}


\subsection{Level 1: Word Interaction Within Sentences}

\begin{equation}
K_{\text{sentence}} = K_{\text{verb}} \times H \times Q \times T
\end{equation}

\textbf{Worked Example:}
\begin{align*}
\text{``X causes Y''} &\rightarrow K = 0.90 \\
\text{``X might cause Y''} &\rightarrow K = 0.90 \times 0.50 = 0.45 \\
\text{``X might sometimes cause Y in some people''} &\rightarrow K \approx 0.10
\end{align*}


\subsection{Level 2: Sentence Interaction Within Paragraphs}

\textbf{Pattern 2.1: Reinforcement} --- $K_{\text{para}} = K_{\text{lead}} + \sum \text{bonus}$

\textbf{Pattern 2.2: Qualification} --- $K_{\text{para}} = K_{\text{lead}} \times \prod \text{factor}$

\textbf{Pattern 2.3: Contradiction} --- $K_{\text{para}} = \frac{K_{S1} + K_{S2}}{2} \times \text{penalty}$

\textbf{Pattern 2.4: Causal Chains} --- $M_{\text{chain}} = \min(M_i) \times 0.9^{n-1}$


\subsection{Cross-Section Consistency Rules}

\begin{enumerate}
    \item \textbf{Abstract $\leq$ Body}: $K_{\text{abstract}} \leq \max(K_{\text{body}})$
    \item \textbf{Title $\leq$ Abstract}: $K_{\text{title}} \leq K_{\text{abstract}}$
    \item \textbf{Conclusion $\leq$ Results}: $K_{\text{conclusion}} \leq K_{\text{results}} \times \text{factor}$
\end{enumerate}

Violations constitute distinct error types: \textbf{Abstract Inflation}, \textbf{Conclusion Creep}, \textbf{Headline Hype}, \textbf{Spin}.

\section{Implementation: The ESL Checker}

\begin{analogybox}[The Spell Checker]
Before spell checkers, proofreading was manual and error-prone. The ESL Checker does for epistemic calibration what spell checkers did for spelling. It automates detection of overclaiming.
\end{analogybox}


\subsection{Architecture}

Three-stage pipeline:
\begin{enumerate}
    \item \textbf{Claim Extraction}: Identify assertive sentences using speech-act classification
    \item \textbf{K-Score Calculation}: Extract linguistic markers across K1--K5
    \item \textbf{M-Score Estimation}: Query reference database or use LLM-based estimation
\end{enumerate}

\textbf{Solution:} Pre-submission ESL check, editorial integration, reviewer training.

\subsection{Media Literacy Education}
ESL can be taught as a practical critical thinking framework.

\subsection{Misinformation Detection: The $\Delta$ Index}
\begin{equation}
\text{Misinfo-Index} = \text{mean}(\Delta) \times \sqrt{\text{OC-Rate}}
\end{equation}
Texts with Misinfo-Index $> 0.30$ are flagged. Pilot study: 87\% correct classification.

\subsection{Policy Communication}
Calibrate policy language to evidence strength:
\begin{itemize}
    \item $M > 0.85$: ``Evidence strongly supports...''
    \item $M \approx 0.60$: ``Current evidence suggests...''
    \item $M \approx 0.40$: ``Preliminary findings indicate...''
    \item $M < 0.25$: ``Limited evidence exists for...''
\end{itemize}

\section{The Methodological Discovery: Causal Models as LLM Activation Structures}

\begin{principlebox}[The Core Methodological Insight]
\textbf{A causal model can serve as an activation structure for latent LLM knowledge.}

The causal model is not something the LLM must \emph{learn}---it is a structure that \emph{organizes and activates} what the LLM already knows.
\end{principlebox}


\subsection{How ESL Demonstrated This Principle}

\begin{enumerate}
    \item We had a causal model: $K \leq M$
    \item We did NOT train the LLM
    \item We used the causal model as system configuration
    \item The LLM's latent knowledge was activated: $r = 0.999$
\end{enumerate}

\begin{principlebox}[A New Paradigm for Domain-Specific AI]

\section{Discussion}

\subsection{What ESL Achieves}

\begin{enumerate}
    \item \textbf{LLMs possess latent epistemic competence}: $r = 0.999$ cross-model agreement
    \item \textbf{System-level activation works}: 94\% reduction in overclaiming
    \item \textbf{Theoretically grounded}: Operationalizes Grice's Maxim of Quality
    \item \textbf{Precisely defined scope}: Four necessary conditions, seven exclusion categories
\end{enumerate}


\subsection{Limitations and Honest Uncertainties}

\begin{enumerate}
    \item \textbf{M-Score availability}: For novel claims, M-Scores must be estimated
    \item \textbf{K-Metric calibration across languages}: Hedging conventions differ
    \item \textbf{Context sensitivity}: Requires human judgment for scope conditions
    \item \textbf{Adversarial robustness}: May be vulnerable to prompt injection
    \item \textbf{Training data dependency}: Cannot activate knowledge that was never learned
\end{enumerate}


\subsection{Future Directions}

\begin{enumerate}
    \item Automated ESL Checking (browser extensions, manuscript tools)
    \item Expanded M-Score Database (medicine, economics, climate science)
    \item Multilingual K-Metrics
    \item Integration with Peer Review
    \item Longitudinal Tracking
\end{enumerate}
